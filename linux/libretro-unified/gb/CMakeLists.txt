cmake_minimum_required(VERSION 3.18)
project(pokedaisy_gb_compat C)
set(THWONP ${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/pokedaisy-libretro)
add_subdirectory(${THWONP}/libretro thwonp)
# A checked build-time adaptation, without a dirty/modified pinned submodule.
set(original ${THWONP}/libretro/pd_libretro.c)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${original})
file(READ ${original} wrapper)
set(before "    struct retro_throttle_state th = {0, 0.0f};\n    bool got = realEnv(RETRO_ENVIRONMENT_GET_THROTTLE_STATE, &th);\n    ff = got && th.mode == RETRO_THROTTLE_FAST_FORWARD;\n    rewinding = got && th.mode == RETRO_THROTTLE_REWINDING;")
string(FIND "${wrapper}" "${before}" found)
if(found LESS 0)
  message(FATAL_ERROR "Review the pinned Thwonp playback hook before updating the compatibility adapter")
endif()
string(REPLACE "${before}" "    struct pd_playback playback = pd_playback_query(realEnv);\n    ff = playback.fast_forward;\n    rewinding = playback.rewinding;" wrapper "${wrapper}")
string(REPLACE "#include \"field_items.h\"" "#include \"pd_playback.h\"\n#include \"field_items.h\"" wrapper "${wrapper}")
set(generated ${CMAKE_CURRENT_BINARY_DIR}/pd_libretro_compat.c)
file(WRITE ${generated} "${wrapper}")
get_target_property(sources pokedaisy_libretro SOURCES)
set(adapted)
foreach(source IN LISTS sources)
  if(source STREQUAL "pd_libretro.c" OR source STREQUAL "${original}")
    continue()
  endif()
  if(NOT IS_ABSOLUTE "${source}" AND NOT source MATCHES "^\\$<")
    set(source ${THWONP}/libretro/${source})
  endif()
  list(APPEND adapted "${source}")
endforeach()
list(APPEND adapted ${generated})
set_property(TARGET pokedaisy_libretro PROPERTY SOURCES "${adapted}")
target_include_directories(pokedaisy_libretro PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${THWONP}/libretro)
