cmake_minimum_required(VERSION 3.18)
project(pokedaisy-pocket C)
set(CMAKE_C_STANDARD 11)
option(POCKET_BUNDLED_MGBA "Build pinned upstream mGBA" ON)
if(POCKET_BUNDLED_MGBA)
  set(BUILD_QT OFF CACHE BOOL "" FORCE)
  set(BUILD_SDL OFF CACHE BOOL "" FORCE)
  set(BUILD_GL OFF CACHE BOOL "" FORCE)
  set(BUILD_GLES2 OFF CACHE BOOL "" FORCE)
  set(BUILD_GLES3 OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(BUILD_STATIC ON CACHE BOOL "" FORCE)
  set(USE_DISCORD_RPC OFF CACHE BOOL "" FORCE)
  set(USE_LUA OFF CACHE BOOL "" FORCE)
  set(USE_FFMPEG OFF CACHE BOOL "" FORCE)
  set(USE_LIBZIP OFF CACHE BOOL "" FORCE)
  set(USE_PNG OFF CACHE BOOL "" FORCE)
  set(USE_SQLITE3 OFF CACHE BOOL "" FORCE)
  set(USE_ELF OFF CACHE BOOL "" FORCE)
  set(USE_LZMA OFF CACHE BOOL "" FORCE)
  set(USE_DEBUGGERS OFF CACHE BOOL "" FORCE)
  add_subdirectory(../third_party/mgba mgba)
  set(MGBA_LIB mgba)
else()
  find_library(MGBA_LIB mgba REQUIRED)
endif()
find_path(SDL2_INCLUDE_DIRS SDL.h PATH_SUFFIXES SDL2 REQUIRED)
find_library(SDL2_LIBRARIES SDL2 REQUIRED)
find_package(Threads REQUIRED)
file(GLOB PD_CORE ../3ds/core/*.c)
add_executable(pokedaisy-pocket main.c polished.c ../app/src/main/cpp/pk_wide_gb.c ${PD_CORE} ../3ds/desktop/pd_png.c)
target_include_directories(pokedaisy-pocket PRIVATE ../3ds/core ../third_party/mgba/include ${CMAKE_CURRENT_BINARY_DIR}/mgba/include ${SDL2_INCLUDE_DIRS})
target_compile_definitions(pokedaisy-pocket PRIVATE _POSIX_C_SOURCE=200809L POCKET_NATIVE)
target_compile_options(pokedaisy-pocket PRIVATE -Wall -Wextra)
target_link_libraries(pokedaisy-pocket ${MGBA_LIB} ${SDL2_LIBRARIES} m Threads::Threads ${CMAKE_DL_LIBS})
option(POCKET_TESTS "Build integration harness (not shipped)" OFF)
if(POCKET_TESTS)
 add_executable(pocket-rom-test test_rom.c ../app/src/main/cpp/pk_wide_gb.c ${PD_CORE} ../3ds/desktop/pd_png.c)
 target_include_directories(pocket-rom-test PRIVATE ../3ds/core ../third_party/mgba/include ${CMAKE_CURRENT_BINARY_DIR}/mgba/include ${SDL2_INCLUDE_DIRS})
 target_compile_definitions(pocket-rom-test PRIVATE _POSIX_C_SOURCE=200809L POCKET_NATIVE)
 target_compile_options(pocket-rom-test PRIVATE -UNDEBUG)
 target_link_libraries(pocket-rom-test ${MGBA_LIB} ${SDL2_LIBRARIES} m Threads::Threads ${CMAKE_DL_LIBS})
 add_executable(pocket-decoder-test test_polished.c ${PD_CORE} ../3ds/desktop/pd_png.c)
 target_include_directories(pocket-decoder-test PRIVATE ../3ds/core)
 target_compile_options(pocket-decoder-test PRIVATE -UNDEBUG)
 target_link_libraries(pocket-decoder-test m)
 add_executable(pocket-runtime-test test_runtime.c polished.c ../app/src/main/cpp/pk_wide_gb.c ${PD_CORE} ../3ds/desktop/pd_png.c)
 target_include_directories(pocket-runtime-test PRIVATE ../3ds/core ../third_party/mgba/include ${CMAKE_CURRENT_BINARY_DIR}/mgba/include ${SDL2_INCLUDE_DIRS})
 target_compile_definitions(pocket-runtime-test PRIVATE _POSIX_C_SOURCE=200809L POCKET_NATIVE)
 target_compile_options(pocket-runtime-test PRIVATE -UNDEBUG)
 target_link_libraries(pocket-runtime-test ${MGBA_LIB} ${SDL2_LIBRARIES} m Threads::Threads ${CMAKE_DL_LIBS})
endif()
